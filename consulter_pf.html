// supabase/functions/super-action/Login.ts // ───────────────────────────────────────────────────────────────────────── // LOGIN — DURCI + ANTI-BRUTEFORCE CÔTÉ SERVEUR // • Plus aucun log du body (mot de passe). // • Message générique (anti-énumération de comptes). // • Verrouillage SERVEUR via la table login_attempts : // - > MAX_FAILS échecs pour un identifiant en WINDOW_MIN min → 429 // - garde-fou IP (anti-spraying) → 429 // - les tentatives sont enregistrées par la fonction (IP réelle) // - une connexion réussie purge les échecs récents de l'identifiant // Le verrou localStorage du front ne sert plus que de confort UX. // // ⚠️ Déployer AVEC le patch login.html (qui n'insère plus dans login_attempts). // ───────────────────────────────────────────────────────────────────────── import { serve } from "https://deno.land/std/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2.48.0"; import bcrypt from "https://esm.sh/bcryptjs@2.4.3"; const URL = Deno.env.get("URL")!; const SERVICE_ROLE_KEY = Deno.env.get("SERVICE_ROLE_KEY")!; const supabase = createClient(URL, SERVICE_ROLE_KEY); const WINDOW_MIN = 15; // fenêtre glissante const MAX_FAILS = 5; // échecs max par identifiant const MAX_FAILS_IP = 20; // échecs max par IP (anti-spraying) const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "authorization, x-client-info, content-type", "Content-Type": "application/json", }; const INVALID = JSON.stringify({ error: "Identifiants invalides" }); const LOCKED = JSON.stringify({ error: "Trop de tentatives. Réessayez dans 15 minutes." }); function clientIp(req: Request): string { const xff = req.headers.get("x-forwarded-for") || ""; const ip = (xff.split(",")[0] || "").trim() || req.headers.get("cf-connecting-ip") || ""; return ip.slice(0, 64); } serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: CORS_HEADERS }); try { const params = new URLSearchParams(await req.text()); // jamais loggé const username = (params.get("username") || "").trim(); const password = params.get("password") || ""; if (!username || !password) { return new Response(JSON.stringify({ error: "Champs manquants" }), { status: 400, headers: CORS_HEADERS }); } const safeUsername = username.replace(/[\\%_]/g, "\\$&"); const ip = clientIp(req); const sinceIso = new Date(Date.now() - WINDOW_MIN * 60000).toISOString(); // ── Anti-bruteforce : par identifiant ── const { count: userFails } = await supabase .from("login_attempts").select("id", { count: "exact", head: true }) .eq("success", false).ilike("username", safeUsername).gte("created_at", sinceIso); if ((userFails ?? 0) >= MAX_FAILS) { return new Response(LOCKED, { status: 429, headers: CORS_HEADERS }); } // ── Anti-bruteforce : par IP (spraying multi-comptes) ── if (ip) { const { count: ipFails } = await supabase .from("login_attempts").select("id", { count: "exact", head: true }) .eq("success", false).eq("ip_hint", ip).gte("created_at", sinceIso); if ((ipFails ?? 0) >= MAX_FAILS_IP) { return new Response(LOCKED, { status: 429, headers: CORS_HEADERS }); } } async function record(success: boolean) { try { await supabase.from("login_attempts").insert({ username, success, ip_hint: ip, created_at: new Date().toISOString(), }); } catch (_) { /* le log ne doit jamais bloquer le login */ } } const { data: user, error } = await supabase .from("users").select("*").ilike("nom", safeUsername).maybeSingle(); if (error) { console.log("Login DB error"); return new Response(JSON.stringify({ error: "Database error" }), { status: 500, headers: CORS_HEADERS }); } if (!user || !user.password_hash || !bcrypt.compareSync(password, user.password_hash)) { await record(false); return new Response(INVALID, { status: 401, headers: CORS_HEADERS }); } // Succès : on enregistre et on purge les échecs récents de cet identifiant await record(true); try { await supabase.from("login_attempts").delete() .eq("success", false).ilike("username", safeUsername).gte("created_at", sinceIso); } catch (_) { /* ignore */ } return new Response( JSON.stringify({ success: true, user: { id: user.id, nom: user.nom, role: user.role, badgeuid: user.badgeuid, site_code: user.site_code, service: user.service, }, }), { status: 200, headers: CORS_HEADERS }, ); } catch (_e) { console.log("Login internal error"); return new Response(JSON.stringify({ error: "Erreur interne" }), { status: 500, headers: CORS_HEADERS }); } });
