// supabase/functions/bright-api/login-nfc.ts // ───────────────────────────────────────────────────────────────────────── // LOGIN NFC — VERSION DURCIE // Correction principale : la version précédente faisait select("*") et // renvoyait TOUT l'utilisateur, y compris password_hash → fuite du hash // à chaque login badge. On ne renvoie plus que les champs nécessaires. // + en-têtes CORS + plus de fuite de e.message. // // ⚠️ À déployer avec le reste (le front lit toujours data.user). // ───────────────────────────────────────────────────────────────────────── import { serve } from "https://deno.land/std/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2.48.0"; const URL = Deno.env.get("URL")!; const SERVICE_ROLE_KEY = Deno.env.get("SERVICE_ROLE_KEY")!; const supabase = createClient(URL, SERVICE_ROLE_KEY); const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "authorization, x-client-info, content-type", "Content-Type": "application/json", }; serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: CORS_HEADERS }); try { const params = new URLSearchParams(await req.text()); const nfc_uid = params.get("nfc_uid"); if (!nfc_uid) { return new Response(JSON.stringify({ error: "Missing NFC UID" }), { status: 400, headers: CORS_HEADERS }); } // Sélection EXPLICITE des champs — jamais password_hash const { data: user, error } = await supabase .from("users") .select("id, nom, role, badgeuid, site_code, service") .eq("badgeuid", nfc_uid) .maybeSingle(); if (error || !user) { return new Response(JSON.stringify({ error: "Badge inconnu" }), { status: 401, headers: CORS_HEADERS }); } return new Response(JSON.stringify({ user }), { status: 200, headers: CORS_HEADERS }); } catch (_e) { return new Response(JSON.stringify({ error: "Erreur interne" }), { status: 500, headers: CORS_HEADERS }); } });
